cmake_minimum_required(VERSION 3.21)
project(JLVoiceAsr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WHISPER_SOURCE)
  message(FATAL_ERROR "Set WHISPER_SOURCE to the pinned whisper.cpp checkout")
endif()
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory("${WHISPER_SOURCE}" whisper)
add_executable(JLVoiceAsr worker.cpp)
if(WIN32)
  enable_language(RC)
  target_sources(JLVoiceAsr PRIVATE version.rc)
endif()
target_include_directories(JLVoiceAsr PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  # Let CMake merge this with MSVC's generated manifest; do not embed it twice.
  target_sources(JLVoiceAsr PRIVATE worker.manifest)
elseif(WIN32)
  target_sources(JLVoiceAsr PRIVATE worker.rc)
endif()
target_link_libraries(JLVoiceAsr PRIVATE whisper)

# Model-free unit tests for the transcript guard (run in CI and locally).
add_executable(JLVoiceGuardTest guard_test.cpp)
target_include_directories(JLVoiceGuardTest PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(JLVoiceGuardTest PRIVATE /utf-8)
endif()
